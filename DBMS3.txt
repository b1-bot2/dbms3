CREATE TABLE PUBLISHER (
     Pub_id INT PRIMARY KEY,
     Name VARCHAR(50) UNIQUE,
     Address VARCHAR(100),
     Phone VARCHAR(15)
);

CREATE TABLE BOOK (
     Book_id INT PRIMARY KEY,
     Title VARCHAR(100),
     Pub_id INT,
     Pub_Year INT,
     FOREIGN KEY (Pub_id) REFERENCES PUBLISHER(Pub_id)
);

CREATE TABLE BOOK_AUTHORS (
     Book_id INT,
     Author_Name VARCHAR(50),
     FOREIGN KEY (Book_id) REFERENCES BOOK(Book_id)
);

CREATE TABLE LIBRARY_BRANCH (
     Branch_id INT PRIMARY KEY,
     Branch_Name VARCHAR(50),
     Address VARCHAR(100)
);

CREATE TABLE BOOK_COPIES (
     Book_id INT,
     Branch_id INT,
     No_of_Copies INT,
     FOREIGN KEY (Book_id) REFERENCES BOOK(Book_id),
     FOREIGN KEY (Branch_id) REFERENCES LIBRARY_BRANCH(Branch_id)
);

CREATE TABLE BOOK_LENDING (
     Book_id INT,
     Branch_id INT,
     Card_No INT,
     Date_Out DATE,
     Due_Date DATE,
     FOREIGN KEY (Book_id) REFERENCES BOOK(Book_id),
     FOREIGN KEY (Branch_id) REFERENCES LIBRARY_BRANCH(Branch_id)
);

-- Query 1 - Book details and copies by branch

SELECT
  B.Book_id,
  B.Title,
  P.Name AS Publisher_Name,
  BA.Author_Name,
  LB.Branch_Name,
  BC.No_of_Copies
FROM BOOK B
JOIN PUBLISHER P ON B.Pub_id = P.Pub_id
JOIN BOOK_AUTHORS BA ON B.Book_id = BA.Book_id
JOIN BOOK_COPIES BC ON B.Book_id = BC.Book_id
JOIN LIBRARY_BRANCH LB ON BC.Branch_id = LB.Branch_id;


-- Query 2 - Borrowers with more than eight loans

SELECT
  Card_No,
  COUNT(*) AS Total_Books_Borrowed
FROM BOOK_LENDING
WHERE Date_Out BETWEEN '2019-01-01' AND '2019-06-30'
GROUP BY Card_No
HAVING COUNT(*) > 8;

-- Query 1 - Customers above Bangalore average

SELECT COUNT(*) AS Count_Above_Avg
FROM CUSTOMER
WHERE Grade > (
     SELECT AVG(Grade)
     FROM CUSTOMER
     WHERE City = 'Bangalore'
);


-- Query 2 - Salesmen with multiple customers

SELECT
     S.Salesman_id,
     S.Name,
COUNT(C.Customer_id) AS No_of_Customers
FROM SALESMAN S
JOIN CUSTOMER C ON S.Salesman_id = C.Salesman_id
GROUP BY S.Salesman_id, S.Name
HAVING COUNT(C.Customer_id) > 1;


-- Query 3 - Same-city customer status using UNION

SELECT S.Salesman_id, S.Name, S.City, 'Has Customer in Same City' AS Status
FROM SALESMAN S
JOIN CUSTOMER C ON S.Salesman_id = C.Salesman_id
WHERE S.City = C.City
UNION
SELECT S.Salesman_id, S.Name, S.City, 'No Customer in Same City' AS Status
FROM SALESMAN S
WHERE S.Salesman_id NOT IN (
     SELECT Salesman_id FROM CUSTOMER WHERE City = S.City
);


-- Query 4 - Daily highest order

CREATE VIEW Highest_Order_Salesman AS
SELECT
     S.Salesman_id,
     S.Name AS Salesman_Name,
     C.Cust_Name AS Customer_Name,
     O.Ord_No,
O.Purchase_Amt,
     O.Ord_Date
FROM SALESMAN S
JOIN ORDERS O ON S.Salesman_id = O.Salesman_id
JOIN CUSTOMER C ON O.Customer_id = C.Customer_id
WHERE O.Purchase_Amt = (
     SELECT MAX(Purchase_Amt)
     FROM ORDERS
     WHERE Ord_Date = O.Ord_Date
);

SELECT * FROM Highest_Order_Salesman;

-- Query 2 - Movies with actors in multiple movies

SELECT DISTINCT M.Mov_Title
FROM MOVIES M
JOIN MOVIE_CAST MC ON M.Mov_id = MC.Mov_id
WHERE MC.Act_id IN (
  SELECT Act_id
  FROM MOVIE_CAST
  GROUP BY Act_id
  HAVING COUNT(DISTINCT Mov_id) >= 2);


-- Query 3 - Actors before 2000 and after 2015

SELECT DISTINCT A.Act_Name
FROM ACTOR A
JOIN MOVIE_CAST MC1 ON A.Act_id = MC1.Act_id
JOIN MOVIES M1 ON MC1.Mov_id = M1.Mov_id
JOIN MOVIE_CAST MC2 ON A.Act_id = MC2.Act_id
JOIN MOVIES M2 ON MC2.Mov_id = M2.Mov_id
WHERE M1.Mov_Year < 2000 AND M2.Mov_Year > 2015;


-- Query 4 - Highest rating per movie

SELECT M.Mov_Title, MAX(R.Rev_Stars) AS Highest_Stars
FROM MOVIES M
JOIN RATING R ON M.Mov_id = R.Mov_id
GROUP BY M.Mov_Title
ORDER BY M.Mov_Title;


-- Query 5 - Set XYZ movie ratings to five

UPDATE RATING
SET Rev_Stars = 5
WHERE Mov_id IN (
    SELECT M.Mov_id
    FROM MOVIES M
    JOIN DIRECTOR D ON M.Dir_id = D.Dir_id
    WHERE D.Dir_Name = 'XYZ' );

SELECT M.Mov_Title, R.Rev_Stars FROM MOVIES M JOIN DIRECTOR D ON D.Dir_id = M.Dir_id JOIN RATING R ON R.Mov_id = M.Mov_id WHERE D.Dir_Name = 'XYZ' ORDER BY M.Mov_Title;

FROM STUDENT S
JOIN CLASS C ON S.USN = C.USN

-- Query 2 - Student counts by semester, section and gender

SELECT SS.Sem, SS.Sec, S.Gender, COUNT(*) AS Total
FROM STUDENT S
JOIN CLASS C ON S.USN = C.USN
JOIN SEMSEC SS ON C.SSID = SS.SSID

ORDER BY SS.Sem, SS.Sec;

-- Query 5 - Eighth semester IA categories

SELECT S.USN, S.SName, SS.Sem, SS.Sec, IA.FinalIA,
CASE
 WHEN IA.FinalIA BETWEEN 17 AND 20 THEN 'Outstanding'
 WHEN IA.FinalIA BETWEEN 12 AND 16 THEN 'Average'
ELSE 'Weak'
END AS Category
FROM STUDENT S
JOIN CLASS C ON S.USN = C.USN
JOIN SEMSEC SS ON C.SSID = SS.SSID
JOIN IA_MARKS IA ON S.USN = IA.USN
WHERE SS.Sem = 8 AND SS.Sec IN ('A','B','C');

-- Query 1 - Projects involving Seo

SELECT DISTINCT P.PNo
FROM PROJECT P
LEFT JOIN WORKS_ON W ON P.PNo = W.PNo
LEFT JOIN EMPLOYEE E ON W.SSN = E.SSN
LEFT JOIN DEPARTMENT D ON P.DNo = D.DNo
WHERE E.Name LIKE '%Seo' OR D.MgrSSN IN (SELECT SSN FROM EMPLOYEE
WHERE Name LIKE '%Seo');


-- Query 2 - Show IoT salaries after a ten percent raise

SELECT E.SSN, E.Name, E.Salary, E.Salary*1.10 AS NewSalary
FROM EMPLOYEE E
JOIN WORKS_ON W ON E.SSN = W.SSN
JOIN PROJECT P ON W.PNo = P.PNo
WHERE P.PName = 'IoT';


-- Query 3 - Accounts salary statistics

SELECT SUM(E.Salary) AS TotalSalary,
    MAX(E.Salary) AS MaxSalary,
    MIN(E.Salary) AS MinSalary,
    AVG(E.Salary) AS AvgSalary
FROM EMPLOYEE E
JOIN DEPARTMENT D ON E.DNo = D.DNo
WHERE D.DName = 'Accounts';


-- Query 4 - Employees on every department 5 project

SELECT E.Name
FROM EMPLOYEE E
WHERE NOT EXISTS (
     SELECT P.PNo
     FROM PROJECT P
     WHERE P.DNo = 5
     AND NOT EXISTS (
         SELECT W.SSN
         FROM WORKS_ON W
         WHERE W.PNo = P.PNo AND W.SSN = E.SSN
     )
);
